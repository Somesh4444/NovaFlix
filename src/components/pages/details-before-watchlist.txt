import { useEffect, useState } from "react";
import { useParams, Link } from "react-router-dom";
import {Star, Play, Clock, Calendar, Globe, Film, ExternalLink, Link2, Heart, Flame, X} from "lucide-react";

export default function MovieDetails() {
  // 1. 🚀 FIXED: Capture BOTH the layout 'type' and the 'id' parameters from your updated Router Switch paths
  const { type, id } = useParams(); 
  
  const [movie, setMovie] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  
  // Modal toggle state and trailer key pointer storage tracking
  const [showModal, setShowModal] = useState(false);
  const [trailerKey, setTrailerKey] = useState("");

  const API_KEY = "1476cff43fbcd08886135a08bf98665e";

  useEffect(() => {
    const fetchMovieDetails = async () => {
      try {
        setLoading(true);
        setError(null); // Reset layout error state states on fresh re-mounts

        // 2. 🚀 FIXED: Swap out the hardcoded '/movie/' sub-path wrapper block segment for your dynamic dynamic variable segment `${type}`
        const response = await fetch(
          `https://api.themoviedb.org/3/${type}/${id}?api_key=${API_KEY}&append_to_response=credits,recommendations,keywords,alternative_titles,videos`
        );

        if (!response.ok) {
          throw new Error("Failed to capture comprehensive TMDB archive records.");
        }

        const data = await response.json();
        setMovie(data);

        // Scan and extract the exact streaming index clip target element path indicator key from YouTube arrays 
        if (data.videos && data.videos.results) {
          const officialTrailer = data.videos.results.find(
            (vid) => vid.type === "Trailer" && vid.site === "YouTube"
          ) || data.videos.results.find((vid) => vid.site === "YouTube");
          
          if (officialTrailer) {
            setTrailerKey(officialTrailer.key);
          } else {
            setTrailerKey("");
          }
        }
      } catch (err) {
        console.error(err);
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };

    // 3. 🚀 FIXED: Add tracking array safeguards for route mutations across related elements
    if (id && type) fetchMovieDetails();
  }, [type, id]); // Re-fire safely if either parameter changes

  if (loading) {
    return (
      <div className="min-h-screen bg-[#020617] flex items-center justify-center text-white">
        <div className="w-8 h-8 border-4 border-cyan-400 border-t-transparent rounded-full animate-spin"></div>
      </div>
    );
  }

  if (error || !movie) {
    return (
      <div className="min-h-screen bg-[#020617] flex items-center justify-center text-red-400 p-4">
        <div className="bg-red-500/10 border border-red-500/20 p-6 rounded-2xl max-w-md text-center">
          <p className="font-bold mb-2">Record Read Error</p>
          <p className="text-sm text-gray-400">{error || "Movie dataset does not exist."}</p>
          <Link to="/" className="text-xs text-cyan-400 underline mt-4 inline-block">Back to Directory</Link>
        </div>
      </div>
    );
  }

  const formatCurrency = (num) => {
    if (!num) return "Not Documented";
    return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD", maximumFractionDigits: 0 }).format(num);
  };

  const formatRuntime = (mins) => {
    if (!mins) return "N/A";
    const hrs = Math.floor(mins / 60);
    const remMins = mins % 60;
    return hrs > 0 ? `${hrs}h ${remMins}m` : `${remMins}m`;
  };

  // 4. 🚀 FIXED: Fallbacks for Crew Directors / Executive Content creators since properties mismatch in TV endpoints
  const director = movie.credits?.crew?.find(member => member.job === "Director") || 
                   movie.created_by?.[0] || { name: "Not Documented" };

  return (
    <section className="min-h-screen bg-[#020617] text-gray-100 py-12 px-6 relative overflow-x-hidden select-none">
      
      {/* 1. HERO BACKDROP IMAGE BANNER OVERLAY */}
      {movie.backdrop_path && (
        <div className="absolute top-0 left-0 w-full h-[60vh] md:h-[75vh] opacity-10 pointer-events-none z-0">
          <img 
            src={`https://image.tmdb.org/t/p/w1280${movie.backdrop_path}`} 
            alt="" 
            className="w-full h-full object-cover"
          />
          <div className="absolute inset-0 bg-linear-to-b from-transparent via-[#020617]/50 to-[#020617]" />
        </div>
      )}

      <div className="max-w-7xl mx-auto grid grid-cols-1 lg:grid-cols-12 gap-8 relative z-10 mt-4">
        
        {/* LEFT COLUMN: POSTER & EXHAUSTIVE METRICS SIDE PANEL */}
        <div className="lg:col-span-4 flex flex-col gap-6 max-w-90 mx-auto lg:mx-0 w-full">
          
          {/* Main Card Element */}
          <div className="relative aspect-2/3 rounded-2xl overflow-hidden border border-white/5 bg-slate-900 shadow-2xl">
            <img 
              src={movie.poster_path ? `https://image.tmdb.org/t/p/w500${movie.poster_path}` : "https://placehold.co/500x750?text=No+Poster+Found"} 
              alt={movie.title || movie.name}
              className="w-full h-full object-cover"
            />
            
            {/* Rating Footer Strip */}
            <div className="absolute bottom-3 left-3 right-3 bg-black/75 backdrop-blur-md p-3 rounded-xl border border-white/10 flex justify-between items-center">
              <div className="flex items-center gap-1.5">
                <Star size={14} fill="#22d3ee" className="text-cyan-400" />
                <span className="font-bold text-sm">{movie.vote_average?.toFixed(1)}/10</span>
              </div>
              <span className="text-[10px] text-gray-400 uppercase tracking-wider font-bold">
                {movie.vote_count?.toLocaleString()} Ratings
              </span>
            </div>
          </div>

          {/* QUICK TRACK STAT CARDS GRID */}
          <div className="grid grid-cols-2 gap-2 text-xs font-semibold text-gray-400">
            <div className="bg-white/5 p-3 rounded-xl border border-white/5 flex items-center gap-2">
              <Clock size={14} className="text-cyan-400 shrink-0" />
              <div>
                <p className="text-[9px] text-gray-500 uppercase font-black">{type === "movie" ? "Runtime" : "Seasons"}</p>
                {/* 🚀 FIXED: TV shows use .number_of_seasons instead of movie runtime properties */}
                <p className="text-gray-200 mt-1">
                  {type === "movie" ? formatRuntime(movie.runtime) : `${movie.number_of_seasons || 1} Seasons`}
                </p>
              </div>
            </div>

            <div className="bg-white/5 p-3 rounded-xl border border-white/5 flex items-center gap-2">
              <Calendar size={14} className="text-cyan-400 shrink-0" />
              <div>
                <p className="text-[9px] text-gray-500 uppercase font-black">Release Date</p>
                {/* 🚀 FIXED: Fallback pattern for tv series release data logs */}
                <p className="text-gray-200 mt-1">{movie.release_date || movie.first_air_date || "N/A"}</p>
              </div>
            </div>

            <div className="bg-white/5 p-3 rounded-xl border border-white/5 flex items-center gap-2">
              <Globe size={14} className="text-cyan-400 shrink-0" />
              <div>
                <p className="text-[9px] text-gray-500 uppercase font-black">Language</p>
                <p className="text-gray-200 uppercase mt-1">{movie.original_language}</p>
              </div>
            </div>

            <div className="bg-white/5 p-3 rounded-xl border border-white/5 flex items-center gap-2">
              <Flame size={14} className="text-cyan-400 shrink-0" />
              <div>
                <p className="text-[9px] text-gray-500 uppercase font-black">Popularity</p>
                <p className="text-gray-200 mt-1">{movie.popularity?.toFixed(0)}</p>
              </div>
            </div>
          </div>

          {/* SUB-DATA SIDE SHEET MODULE */}
          <div className="bg-white/5 border border-white/5 p-4 rounded-xl flex flex-col gap-3 text-xs text-gray-400">
            <h4 className="text-[10px] font-black uppercase text-cyan-400 tracking-wider mb-1">Logistics Record</h4>
            
            {movie.status && (
              <div className="flex justify-between border-b border-white/5 pb-1.5">
                <span>Production Status:</span>
                <span className="text-gray-200 font-bold">{movie.status}</span>
              </div>
            )}
            
            {movie.spoken_languages?.length > 0 && (
              <div className="flex justify-between border-b border-white/5 pb-1.5">
                <span>Spoken Tracks:</span>
                <span className="text-gray-200 font-bold truncate max-w-40">
                  {movie.spoken_languages.map(l => l.english_name).join(", ")}
                </span>
              </div>
            )}

            {movie.production_countries?.length > 0 && (
              <div className="flex justify-between border-b border-white/5 pb-1.5">
                <span>Origin Regions:</span>
                <span className="text-gray-200 font-bold truncate max-w-40">
                  {movie.production_countries.map(c => c.name).join(", ")}
                </span>
              </div>
            )}

            {movie.homepage && (
              <a 
                href={movie.homepage} 
                target="_blank" 
                rel="noreferrer" 
                className="flex items-center justify-center gap-1.5 mt-1 bg-cyan-500/10 hover:bg-cyan-500/20 border border-cyan-400/20 text-cyan-400 py-2 rounded-lg font-bold transition"
              >
                <ExternalLink size={12} /> Launch Official Site
              </a>
            )}
          </div>
        </div>

        {/* RIGHT COLUMN: DETAIL STREAMS */}
        <div className="lg:col-span-8 flex flex-col gap-6">
          
          {/* Header Metadata block */}
          <div>
            <div className="flex flex-wrap items-center gap-2">
              <span className="text-cyan-400 text-[10px] font-black uppercase tracking-widest bg-cyan-500/10 border border-cyan-500/20 px-2.5 py-1 rounded-md">
                {type === "movie" ? "Directed By: " : "Created By: "}<span className="text-cyan-400">{director.name}</span>
              </span>
              {movie.imdb_id && (
                <a 
                  href={`https://www.imdb.com/title/${movie.imdb_id}`}
                  target="_blank"
                  rel="noreferrer"
                  className="text-amber-400 text-[10px] font-black uppercase tracking-widest bg-amber-500/10 border border-amber-500/20 px-2.5 py-1 rounded-md flex items-center gap-1 hover:bg-amber-500/20 transition"
                >
                  IMDb Link <Link2 size={10} />
                </a>
              )}
            </div>
            
            {/* 🚀 FIXED: Dynamic fallbacks to display title for Movies and name for Series */}
            <h1 className="text-3xl md:text-5xl font-black text-white uppercase tracking-wide mt-3">
              {movie.title || movie.name}
            </h1>
            {(movie.original_title || movie.original_name) !== (movie.title || movie.name) && (
              <p className="text-gray-500 text-xs font-bold mt-3">
                Alt Title: {movie.original_title || movie.original_name}
              </p>
            )}
            {movie.tagline && <p className="text-cyan-400/70 italic text-sm mt-1.5">"{movie.tagline}"</p>}
          </div>

          {/* GENRE METRICS CHIPS */}
          <div className="flex flex-wrap gap-2">
            {movie.genres?.map((genre) => (
              <span key={genre.id} className="text-xs font-bold bg-white/5 border border-white/5 px-3 py-1.5 rounded-xl text-gray-300 backdrop-blur-sm">
                {genre.name}
              </span>
            ))}
          </div>

          {/* DYNAMIC TRAILER INTEGRATION ACTION CONTROLS */}
          <div className="flex flex-wrap gap-3 items-center mt-2">
            {trailerKey ? (
              <button 
                onClick={() => setShowModal(true)}
                className="flex items-center gap-2 bg-cyan-400 hover:bg-cyan-300 text-[#020617] px-6 py-3 rounded-xl font-bold text-xs uppercase tracking-wider transition duration-300 scale-100 hover:scale-[1.02] shadow-xl shadow-cyan-400/10 cursor-pointer"
              >
                <Play size={14} fill="#020617" /> Watch Official Trailer
              </button>
            ) : (
              <button 
                disabled
                className="flex items-center gap-2 bg-gray-800 text-gray-500 px-6 py-3 rounded-xl font-bold text-xs uppercase tracking-wider opacity-60 cursor-not-allowed"
              >
                <Play size={14} /> Trailer Unavailable
              </button>
            )}

            {/* STATIC WATCHLIST BUTTON */}
            <button 
              onClick={() => alert("Added to Watchlist! (Static placeholder)")}
              className="flex items-center gap-2 bg-white/5 hover:bg-white/10 border border-white/10 text-gray-200 px-6 py-3 rounded-xl font-bold text-xs uppercase tracking-wider transition duration-300 scale-100 hover:scale-[1.02] cursor-pointer"
            >
              <Heart size={14} className="text-rose-500" /> + Add to Watchlist
            </button>
          </div>

          {/* MOVIE ABSTRACT OVERVIEW */}
          <div>
            <h3 className="text-xs uppercase text-cyan-400 font-bold tracking-widest mb-2">Overview</h3>
            <p className="text-sm text-gray-300 leading-relaxed bg-white/5 border border-white/5 p-4 rounded-xl shadow-inner">
              {movie.overview || "No conceptual directory catalog abstract overview submitted for this ledger index entry."}
            </p>
          </div>

          {/* FINANCES & AUDIENCE METRICS (Only visible for Movies) */}
          {type === "movie" && (
            <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
              <div className="bg-white/5 border border-white/5 p-4 rounded-xl">
                <p className="text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1">Allocated Budget</p>
                <p className="text-base font-black text-white">{formatCurrency(movie.budget)}</p>
              </div>
              <div className="bg-white/5 border border-white/5 p-4 rounded-xl">
                <p className="text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1">Gross Revenue Outflow</p>
                <p className="text-base font-black text-white">{formatCurrency(movie.revenue)}</p>
              </div>
              <div className="bg-white/5 border border-white/5 p-4 rounded-xl">
                <p className="text-[10px] font-bold text-gray-500 uppercase tracking-wider mb-1">Net ROI Ratio</p>
                <p className={`text-base font-black ${movie.revenue - movie.budget >= 0 ? "text-emerald-400" : "text-rose-400"}`}>
                  {movie.revenue && movie.budget ? formatCurrency(movie.revenue - movie.budget) : "N/A"}
                </p>
              </div>
            </div>
          )}

           {/* UPGRADED COMPACT PORTRAIT CAST MEMBERS ROW COMPONENT - TOP 5 */}
            <div className="flex flex-col gap-3">
              <div className="flex justify-between items-center">
                <h3 className="text-xs uppercase text-cyan-400 font-bold tracking-widest">
                  Top Cast
                </h3>
                <span className="text-[10px] font-bold text-gray-500 uppercase tracking-wider">
                  {movie.credits?.cast?.length || 0} Total
                </span>
              </div>

              <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-5 gap-3">
                {movie.credits?.cast?.slice(0, 5).map((actor) => (
                  <div 
                    key={actor.id} 
                    className="bg-linear-to-b from-white/5 to-transparent border border-white/5 rounded-xl overflow-hidden flex flex-col"
                  >
                    <div className="aspect-2/3 w-full overflow-hidden bg-slate-900 border-b border-white/5">
                      <img 
                        src={actor.profile_path ? `https://image.tmdb.org/t/p/w185${actor.profile_path}` : "https://placehold.co/185x278?text=No+Photo"} 
                        alt={actor.name}
                        className="w-full h-full object-cover"
                      />
                    </div>

                    <div className="p-2 flex flex-col justify-center grow min-w-0 bg-[#020617]/20">
                      <p className="text-[14px] font-bold text-gray-200 truncate w-full">
                        {actor.name}
                      </p>
                      <p className="text-[11px] text-cyan-400/70 font-semibold truncate w-full mt-0.5 tracking-wide">
                        {actor.character || "N/A"}
                      </p>
                    </div>
                  </div>
                ))}
              </div>
            </div>
          

          {/* PRODUCTION LOGO NETWORKS */}
          {movie.production_companies?.length > 0 && (
            <div>
              <h3 className="text-xs uppercase text-cyan-400 font-bold tracking-widest mb-3">Production Ecosystem</h3>
              <div className="flex flex-wrap gap-3 items-center bg-white/5 border border-white/5 p-4 rounded-xl">
                {movie.production_companies.map((company) => (
                  <div key={company.id} className="flex items-center gap-2 bg-[#020617]/40 px-3 py-1.5 rounded-lg border border-white/5 text-[11px] font-semibold text-gray-300">
                    {company.logo_path ? (
                      <img 
                        src={`https://image.tmdb.org/t/p/w92${company.logo_path}`} 
                        alt="" 
                        className="h-4 object-contain brightness-200 contrast-200 invert"
                      />
                    ) : (
                      <Film size={11} className="text-cyan-400" />
                    )}
                    <span className="text-[12px]">{company.name}</span>
                  </div>
                ))}
              </div>
            </div>
          )}

          {/* KEYWORDS SEGMENT CHIPS */}
          {movie.keywords?.keywords?.length > 0 && (
            <div>
              <h3 className="text-xs uppercase text-cyan-400 font-bold tracking-widest mb-2.5">Keywords</h3>
              <div className="flex flex-wrap gap-1.5">
                {movie.keywords.keywords.slice(0, 10).map((tag) => (
                  <span key={tag.id} className="text-[13px] font-bold bg-white/5 border border-white/5 px-2 py-1 rounded-md text-gray-400 hover:text-white transition">
                    #{tag.name.replace(/\s+/g, "")}
                  </span>
                ))}
              </div>
            </div>
          )}

           {/* DYNAMIC RECOMMENDATIONS CONTEXT PANEL */}
            {movie.recommendations?.results?.length > 0 && (
              <div className="mt-6 pt-6 border-t border-white/5">
                <div className="flex justify-between items-center mb-4">
                  <h3 className="text-xs uppercase text-cyan-400 font-bold tracking-widest">
                    you might also like...
                  </h3>
                  <span className="text-[10px] font-bold text-gray-500 uppercase tracking-wider">
                    Suggestions
                  </span>
                </div>

                <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
                  {movie.recommendations.results.slice(0, 4).map((rec) => {
                    // 🚀 FIXED: Dynamic route detection logic inside the recommendation loops
                    const itemType = rec.media_type || type;
                    return (
                      <div key={rec.id} className="group flex flex-col gap-2 w-full">
                        <div className="relative aspect-2/3 w-full rounded-xl overflow-hidden bg-slate-900 border border-white/5 shadow-md">
                          <img 
                            src={rec.poster_path ? `https://image.tmdb.org/t/p/w500${rec.poster_path}` : "https://placehold.co/500x750?text=No+Poster+Found"} 
                            alt={rec.title || rec.name}
                            loading="lazy"
                            className="w-full h-full object-cover transition-transform duration-500 group-hover:scale-105"
                          />
                          <div className="absolute inset-0 bg-linear-to-t from-black/80 via-black/10 to-transparent pointer-events-none" />

                          <div className="absolute top-2 right-2 bg-black/60 backdrop-blur-md px-1.5 py-0.5 rounded-lg border border-white/10 flex items-center gap-0.5 z-10">
                            <Star size={9} fill="#22d3ee" className="text-cyan-400" />
                            <span className="text-[10px] font-bold text-gray-200">
                              {rec.vote_average?.toFixed(1) || "0.0"}
                            </span>
                          </div>

                          <div className="absolute top-2 left-2 bg-cyan-500/10 backdrop-blur-md border border-cyan-400/30 text-cyan-400 text-[9px] font-black tracking-wider px-2 py-0.5 rounded-md uppercase z-10">
                            {rec.vote_average >= 7.5 ? "Blockbuster" : "Trending"}
                          </div>

                          <div className="absolute inset-0 flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity duration-300 z-20">
                            {/* 🚀 FIXED: Use the unified details page template mapping parameters */}
                            <Link
                              to={`/details/${itemType}/${rec.id}`}
                              className="w-10 h-10 rounded-full bg-cyan-400 text-black flex items-center justify-center shadow-lg shadow-cyan-400/30 transform scale-90 group-hover:scale-100 transition-transform duration-300 cursor-pointer"
                            >
                              <Play size={16} fill="black" className="ml-0.5" />
                            </Link>
                          </div>
                        </div>

                        <div className="min-w-0 px-0.5">
                          {/* 🚀 FIXED: Use correct target layouts down here as well */}
                          <Link to={`/details/${itemType}/${rec.id}`}>
                            <h4 className="text-xs font-bold text-gray-200 hover:text-cyan-400 transition-colors duration-300 truncate w-full">
                              {rec.title || rec.name}
                            </h4>
                          </Link>
                          <p className="text-[10px] text-gray-500 font-bold mt-0.5 uppercase tracking-wider">
                            {itemType === "movie" ? "Movie" : "Series"} • {rec.release_date || rec.first_air_date ? (rec.release_date || rec.first_air_date).split("-")[0] : "N/A"}
                          </p>
                        </div>
                      </div>
                    );
                  })}
                </div>
              </div>
            )}

        </div>
      </div>

      {/* 2. VIDEO STREAM DIALOGUE POPUP MODAL SCREEN */}
      {showModal && trailerKey && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/80 backdrop-blur-md transition-opacity duration-300 animate-fadeIn">
          <div className="relative w-full max-w-4xl aspect-video bg-[#020617] rounded-2xl overflow-hidden border border-white/10 shadow-2xl shadow-cyan-500/5">
            <div className="absolute top-3 right-3 z-10">
              <button 
                onClick={() => setShowModal(false)}
                className="p-2 rounded-full bg-black/60 hover:bg-black/90 text-white hover:text-cyan-400 border border-white/10 transition cursor-pointer"
              >
                <X size={18} />
              </button>
            </div>

            <iframe
              src={`https://www.youtube.com/embed/${trailerKey}?autoplay=1`}
              title="Movie Trailer Player Screen Frame"
              className="w-full h-full"
              allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture"
              allowFullScreen
            ></iframe>
          </div>
        </div>
      )}

    </section>
  );
}
